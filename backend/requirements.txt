# Walking-skeleton dependencies only.
# The full stack (LangChain/LangGraph, ChromaDB, SQLAlchemy, Telegram,

# fastapi/uvicorn/starlette are left unpinned on the minor level, same
# reasoning as pydantic below: an old exact pin (this one dated to
# mid-2024) drifts out of sync with the ecosystem — installing the MCP
# SDK pulled in a much newer starlette that our old fastapi pin flatly
# didn't support (Router.__init__() no longer takes on_startup). Letting
# pip resolve current compatible versions avoids fighting that by hand.
fastapi>=0.115
uvicorn[standard]>=0.30
# pydantic/pydantic-settings are left unpinned on the patch/minor level
# (>=, not ==) because pydantic-core is a compiled Rust extension: an old
# exact pin has no prebuilt wheel for newer Python versions (e.g. 3.14 on
# Windows) and falls back to a from-source build requiring a Rust+MSVC
# toolchain. Letting pip resolve the latest 2.x avoids that entirely.
pydantic>=2.9,<3
pydantic-settings>=2.5,<3
python-dotenv==1.0.1
httpx==0.27.0
pyyaml==6.0.2

# Memory (Echo): SQLite long-term store + ChromaDB documentary store.
# Left unpinned on the minor level like pydantic above — both have
# compiled dependencies (SQLAlchemy's optional C extension, ChromaDB's
# onnxruntime/numpy stack) and an old exact pin risks the same
# no-prebuilt-wheel problem on newer Python/OS combinations.
sqlalchemy>=2.0
chromadb>=0.5

# MCP server exposing Aegis/Atlas/Echo/Kronos as tools for Hermes Agent
# (or any other MCP client) to call. Official Anthropic SDK.
mcp>=1.0

# Document ingestion (§13) — the front end of the documentary workflow.
# Both are pure Python (no system binary, unlike an OCR stack) and both
# are imported lazily inside their extractor, so an install without them
# still handles the plain-text family and fails on PDF/DOCX with an
# actionable message rather than at import time.
# Deliberately NOT added: pytesseract/Pillow for OCR. This project already
# has a vision model (gemma4, via analyze_image) that reads diagrams and
# screenshots, not just glyphs — a better answer here than a tesseract
# system dependency.
pypdf>=5.0
python-docx>=1.1

# Testing
pytest==8.3.2
pytest-asyncio==0.24.0
